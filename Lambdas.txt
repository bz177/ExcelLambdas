        Adr_2; MAP(Matrix; LAMBDA(x; ADRESSE(ZEILE(x); SPALTE(x); 4)));
        ZUSPALTE(
            WENN(
                UND(
                    WURDEAUSGELASSEN(Zeilenüberschrift);
                    WURDEAUSGELASSEN(Spaltenüberschrift)
                );
                Adr_2 & TR & Matrix;
                Adr_1 & TR & ZÜ_1 & ZÜ_2 & TR & SÜ_1 & SÜ_2 & TR & Matrix
            )
        )
    )
)

            StrS - Skip;
            Richtung = 4;

            WAHR;

        Breite; RightS - LeftS + 1;

KartenWerteImUmkreis
=LAMBDA(
    Start;
    Matrix;
    [Radius];
    [Diagonal_Nein0_Ja1];
    [MitStart_Nein0_Ja1];
    [Matrix0_Sum1_Mitt2_Min3_Max4];
    LET(
        MSM; WENNS(
            WURDEAUSGELASSEN(Matrix0_Sum1_Mitt2_Min3_Max4);
            0;
            UND(Matrix0_Sum1_Mitt2_Min3_Max4 > 0; Matrix0_Sum1_Mitt2_Min3_Max4 < 5);
            Matrix0_Sum1_Mitt2_Min3_Max4;
            WAHR;
            0
        );
        R; WENN(WURDEAUSGELASSEN(Radius); 1; Radius);
        D; WENN(WURDEAUSGELASSEN(Diagonal_Nein0_Ja1); 0; Diagonal_Nein0_Ja1);
        MS; WENN(WURDEAUSGELASSEN(MitStart_Nein0_Ja1); 1; MitStart_Nein0_Ja1);
        sRef; INDIREKT(Start);
        sRow; ZEILE(sRef);
        sCol; SPALTE(sRef);
        mTop; MIN(ZEILE(Matrix));
        mLeft; MIN(SPALTE(Matrix));
        mBottom; MAX(ZEILE(Matrix));
        mRight; MAX(SPALTE(Matrix));
        rows; SEQUENZ(2 * R + 1; 1; sRow - R);
        cols; SEQUENZ(1; 2 * R + 1; sCol - R);
        gridRows; rows + 0 * cols;
        gridCols; 0 * rows + cols;
        dRow; gridRows - sRow;
        dCol; gridCols - sCol;
        distRow; ABS(dRow);
        distCol; ABS(dCol);
        cheby; MAX(distRow; distCol);
        isStart; (dRow = 0) * (dCol = 0);
        allowed; WENN(
            D = 1;
            ((cheby >= 1) * (cheby <= R)) + isStart;
            (
                (dRow = 0) * (distCol >= 1) * (distCol <= R) +
                    (dCol = 0) * (distRow >= 1) * (distRow <= R) + isStart
            )
        );
        allowed2; WENN(MS = 1; allowed; allowed * (1 - isStart));
        inMatrix; (gridRows >= mTop) * (gridRows <= mBottom) * (gridCols >= mLeft) *
            (gridCols <= mRight);
        result; WENN(
            allowed2 * inMatrix;
            INDEX(Matrix; gridRows - mTop + 1; gridCols - mLeft + 1);
            ""
        );
        WENNS(
            MSM = 0;
            result;
            MSM = 1;
            SUMME(result);
            MSM = 2;
            MITTELWERT(result);
            MSM = 3;
            MIN(result);
            MSM = 4;
            MAX(result)
        )
    )
)
